# pydantic/pydantic-settings were exact-pinned (==2.9.2 / ==2.5.2) before
# this project used google-genai's Live API -- google-genai>=2.21.0 itself
# requires pydantic>=2.12.5, so an exact ==2.9.2 pin makes pip's resolver
# report a hard conflict (confirmed with `pip install --dry-run` against
# this exact file: relaxing to the ranges below resolves cleanly to
# pydantic 2.13.5 / pydantic-settings 2.15.0, with fastapi/websockets
# unaffected). Range instead of another exact pin so a future google-genai
# bump doesn't immediately re-break this the same way.
fastapi==0.115.0
uvicorn[standard]==0.30.6
pydantic>=2.12.5,<3.0.0
pydantic-settings>=2.6.0,<3.0.0
numpy==1.26.4
websockets==13.1
python-dotenv==1.0.1
